Inheritance : 
    -- Accessing one class props inside another.
    -- To achieve code re-useability 
    -- By extending one class with another we will use Inheritance.
    syntax : 
        //parent class 
        class A:
            //code 
        
        //child class 
        class B(A):
            //code 
    -- To use the code of class A inside class B. 
    -- class providing data is called as super class or parent class.
    -- class consuming data is called as sub class or child class.
